if(TPGP_ENABLE_MPFR)
  find_path(MPFR_INCLUDE_DIR mpfr.h HINTS /opt/homebrew/opt/mpfr/include /usr/local/include)
  find_library(MPFR_LIBRARY mpfr HINTS /opt/homebrew/opt/mpfr/lib)
  find_path(GMP_INCLUDE_DIR gmp.h HINTS /opt/homebrew/opt/gmp/include /usr/local/include)
  find_library(GMP_LIBRARY gmp HINTS /opt/homebrew/opt/gmp/lib)
  if(MPFR_INCLUDE_DIR AND MPFR_LIBRARY AND GMP_INCLUDE_DIR AND GMP_LIBRARY)
    target_compile_definitions(tpgp_core_rev PUBLIC TPGP_HAVE_MPFR=1)
    target_include_directories(tpgp_core_rev PUBLIC ${MPFR_INCLUDE_DIR} ${GMP_INCLUDE_DIR})
    target_link_libraries(tpgp_core_rev PUBLIC ${MPFR_LIBRARY} ${GMP_LIBRARY})
  else()
    message(STATUS "MPFR not found: validated mode unavailable")
  endif()
endif()
