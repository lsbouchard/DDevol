cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(tpgp_cuda_smoke_rev LANGUAGES C)
  enable_testing()
  set(TPGP_CUDA_STANDALONE ON)
endif()
option(TPGP_ENABLE_CUDA "Build the optional CUDA binary64 branch backend" OFF)
option(TPGP_CUDA_REQUIRE_DEVICE_TEST "Fail standalone CUDA tests when no GPU is usable" OFF)
find_package(Threads REQUIRED)
if(TPGP_ENABLE_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(NOT CMAKE_CUDA_COMPILER)
    message(FATAL_ERROR "TPGP_ENABLE_CUDA=ON requires a CUDA compiler; no CPU fallback was substituted.")
  endif()
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    # Native code plus forward-compatible PTX. Override with the measured
    # device architecture, e.g. 120 for RTX Blackwell, for a native build.
    set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "CUDA architectures, including PTX")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(tpgp_cuda_rev STATIC ../src/cuda_rev.cu)
  set_target_properties(tpgp_cuda_rev PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON
                        CUDA_RUNTIME_LIBRARY Shared CUDA_SEPARABLE_COMPILATION OFF)
  target_compile_options(tpgp_cuda_rev PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--ftz=false;--prec-div=true;--prec-sqrt=true;-Xcompiler=-fno-fast-math,-ffp-contract=off,-Wall,-Wextra>")
  target_link_libraries(tpgp_cuda_rev PUBLIC CUDA::cudart)
  target_compile_definitions(tpgp_cuda_rev PUBLIC TPGP_HAVE_CUDA=1)
else()
  add_library(tpgp_cuda_rev STATIC stub_rev.c)
  target_compile_features(tpgp_cuda_rev PUBLIC c_std_11)
endif()
target_include_directories(tpgp_cuda_rev PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../include")
target_link_libraries(tpgp_cuda_rev PUBLIC Threads::Threads)

if(TPGP_CUDA_STANDALONE)
  add_executable(test_cuda_rev ../tests/test_cuda_rev.c)
  target_compile_features(test_cuda_rev PRIVATE c_std_11)
  target_compile_options(test_cuda_rev PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
  target_link_libraries(test_cuda_rev PRIVATE tpgp_cuda_rev m)
  if(TPGP_CUDA_REQUIRE_DEVICE_TEST)
    add_test(NAME cuda_binary64_rev COMMAND test_cuda_rev --require-device)
  else()
    add_test(NAME cuda_binary64_rev COMMAND test_cuda_rev)
    set_tests_properties(cuda_binary64_rev PROPERTIES SKIP_RETURN_CODE 77)
  endif()
endif()
