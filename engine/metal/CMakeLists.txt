# Can be included from the engine or configured alone for the device smoke test.
cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(tpgp_metal_smoke_rev LANGUAGES C)
  enable_testing()
  set(TPGP_METAL_STANDALONE ON)
endif()

option(TPGP_ENABLE_METAL "Build the Apple Metal packed-key backend" ON)
if(APPLE AND TPGP_ENABLE_METAL)
  enable_language(OBJC)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/pauli_keys_rev.metal" TPGP_METAL_SHADER)
  string(REPLACE "\\" "\\\\" TPGP_METAL_SHADER "${TPGP_METAL_SHADER}")
  string(REPLACE "\"" "\\\"" TPGP_METAL_SHADER "${TPGP_METAL_SHADER}")
  string(REPLACE "\n" "\\n\"\n\"" TPGP_METAL_SHADER "${TPGP_METAL_SHADER}")
  configure_file(shader_source_rev.h.in shader_source_rev.h @ONLY)
  add_library(tpgp_metal_rev STATIC ../src/metal_rev.m)
  target_include_directories(tpgp_metal_rev PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_options(tpgp_metal_rev PRIVATE -fobjc-arc -Wall -Wextra -Wpedantic)
  target_link_libraries(tpgp_metal_rev PUBLIC "-framework Metal" "-framework Foundation")
else()
  add_library(tpgp_metal_rev STATIC ../src/metal_stub_rev.c)
endif()
target_include_directories(tpgp_metal_rev PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../include")
target_compile_features(tpgp_metal_rev PUBLIC c_std_11)
find_package(Threads REQUIRED)
target_link_libraries(tpgp_metal_rev PUBLIC Threads::Threads)

if(TPGP_METAL_STANDALONE)
  add_executable(test_metal_rev ../tests/test_metal_rev.c)
  target_link_libraries(test_metal_rev PRIVATE tpgp_metal_rev)
  add_test(NAME metal_exact_integer_transform COMMAND test_metal_rev)
  set_tests_properties(metal_exact_integer_transform PROPERTIES SKIP_RETURN_CODE 77)
endif()
